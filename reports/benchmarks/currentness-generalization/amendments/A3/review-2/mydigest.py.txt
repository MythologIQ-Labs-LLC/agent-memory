"""Independent recomputation of COMPARISON_RULE_SHA256 from git blobs; no checker import, no inspect."""
import ast, hashlib, re, subprocess, sys
REPO = sys.argv[1]
def src_at(rev): return subprocess.run(["git", "-C", REPO, "show", f"{rev}:scripts/check_generalization_corpus.py"], capture_output=True, text=True, check=True).stdout
def digest(src):
    lines = src.splitlines(keepends=True); tree = ast.parse(src)
    fdefs = {n.name: n for n in tree.body if isinstance(n, ast.FunctionDef)}
    parts = []
    for name in ("tokens", "similarity", "record_texts", "overlap_rule"):
        n = fdefs[name]; start = (n.decorator_list[0].lineno if n.decorator_list else n.lineno) - 1
        parts.append(ast.dump(ast.parse("".join(lines[start:n.end_lineno]))))
    consts = {n.targets[0].id: n.value for n in tree.body if isinstance(n, ast.Assign) and len(n.targets) == 1 and isinstance(n.targets[0], ast.Name)}
    tok = consts["TOKEN"]; assert ast.unparse(tok) == "re.compile('[a-z0-9]+')", ast.unparse(tok)
    thr = ast.literal_eval(consts["THRESHOLD"])
    parts += [tok.args[0].value, repr(thr), f"flags={re.compile(tok.args[0].value).flags}"]
    sites = []
    def walk(node, path):
        if type(node) is ast.Call and type(node.func) is ast.Name and node.func.id == "overlap_rule":
            sites.append(" > ".join(path + [ast.dump(node)]))
        h = None
        if type(node) in (ast.For, ast.AsyncFor): h = "for " + ast.dump(node.target) + " in " + ast.dump(node.iter)
        elif type(node) in (ast.If, ast.While): h = type(node).__name__.lower() + " " + ast.dump(node.test)
        elif type(node) is ast.comprehension: h = "comp " + ast.dump(node.iter) + " if " + "".join(ast.dump(t) for t in node.ifs)
        elif type(node) in (ast.FunctionDef, ast.AsyncFunctionDef): h = "def " + node.name
        for f in node._fields:
            v = getattr(node, f, None)
            for c in (v if isinstance(v, list) else [v]):
                if isinstance(c, ast.AST): walk(c, path + [h] if h else path)
    walk(tree, [])
    b = sum(1 for n in ast.walk(fdefs["check"]) if isinstance(n, ast.Name) and n.id == "templates" and isinstance(n.ctx, ast.Store))
    parts += sites + [f"templates bindings in check: {b}"]
    return hashlib.sha256("\n".join(parts).encode()).hexdigest(), len(sites), b
rec = re.search(r'COMPARISON_RULE_SHA256 = "([0-9a-f]+)"', src_at("656bbe4d")).group(1)
print("recorded at 656bbe4d:", rec)
for rev in ("5555504", "76a9e30", "7b5f7a1", "0173adf7", "656bbe4d"):
    d, ns, b = digest(src_at(rev)); print(rev, d, f"sites={ns} templates_bindings={b}", "MATCH" if d == rec else "differs")
