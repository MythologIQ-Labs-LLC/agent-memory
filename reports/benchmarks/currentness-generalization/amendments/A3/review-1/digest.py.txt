"""Independent recomputation of COMPARISON_RULE_SHA256 from git blobs (no import, no inspect)."""
import ast, hashlib, subprocess
REPO = "/home/user/agent-memory"
def digest(rev, names=("tokens", "similarity", "record_texts", "overlap_rule")):
    src = subprocess.run(["git", "-C", REPO, "show", f"{rev}:scripts/check_generalization_corpus.py"], capture_output=True, text=True, check=True).stdout
    tree = ast.parse(src)
    fns = {n.name: n for n in tree.body if isinstance(n, ast.FunctionDef)}
    missing = [n for n in names if n not in fns]
    if missing:
        return f"missing {missing}"
    parts = [ast.dump(ast.Module(body=[fns[n]], type_ignores=[])) for n in names]
    consts = {}
    for n in tree.body:
        if isinstance(n, ast.Assign) and isinstance(n.targets[0], ast.Name) and n.targets[0].id in ("TOKEN", "THRESHOLD"):
            consts[n.targets[0].id] = n.value
    tok = consts["TOKEN"]; assert ast.unparse(tok.func) == "re.compile" and len(tok.args) == 1 and not tok.keywords, ast.unparse(tok)
    parts += [tok.args[0].value, repr(ast.literal_eval(consts["THRESHOLD"]))]
    return hashlib.sha256("\n".join(parts).encode()).hexdigest() + f"  TOKEN={ast.unparse(tok)} THRESHOLD={ast.unparse(consts['THRESHOLD'])}"
for rev in ("7b5f7a1", "5555504", "76a9e30", "f83edbb4", "0173adf7"):
    print(rev, "4-fn:", digest(rev))
    print(rev, "2-fn:", digest(rev, ("tokens", "similarity")))
    src = subprocess.run(["git", "-C", REPO, "show", f"{rev}:scripts/check_generalization_corpus.py"], capture_output=True, text=True).stdout
    calls = [ast.unparse(n) for n in ast.walk(ast.parse(src)) if isinstance(n, ast.Call) and getattr(n.func, "id", "") == "overlap_rule"]
    print(rev, "overlap_rule call sites:", calls)
