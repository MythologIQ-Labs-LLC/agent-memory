def check(src, strict):
    code = (f"import importlib.util as u, pathlib; s=u.spec_from_file_location('c', {str(CK)!r}); c=u.module_from_spec(s); s.loader.exec_module(c)\n"
            f"c.ROOT=pathlib.Path({str(R)!r})\n"
            f"try: print('ACCEPT', c.template_basis(pathlib.Path({str(src)!r}), require_frozen_blob={strict})[1])\nexcept SystemExit as e: print('refuse', str(e)[:140])")
    return subprocess.run([sys.executable, "-I", "-c", code], capture_output=True, text=True).stdout.strip()
base = S / "gen"; shutil.rmtree(base, ignore_errors=True)
for label, data in (("frozen LF", FROZEN), ("frozen CRLF", FROZEN.replace(b"\n", b"\r\n")), ("audited LF", AUDITED), ("audited CRLF", AUDITED.replace(b"\n", b"\r\n")),
